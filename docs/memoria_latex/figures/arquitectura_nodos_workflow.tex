    },

        minimum width=10.7cm,
        minimum height=0.85cm,

    },
    arrow/.style={-{Latex[length=2.4mm]}, very thick, draw=gray!78!black},

    statearrow/.style={-{Latex[length=1.8mm]}, dotted, thick, draw=gray!65}
]

\node[main=gray] (input) {\textbf{Entrada estructurada}\\consulta, tickers, fechas y rutas CSV};
\node[main=cyan, below=of input] (ingest) {\textbf{1. Validaci\'on m\'inima}\\formato, CSV, fechas y campos b\'asicos};
\node[main=teal, below=of ingest] (analysis) {\textbf{2. LLM central / an\'alisis}\\comprende la consulta, identifica intenci\'on y dise\~na el an\'alisis};
\node[main=orange, below=of analysis] (codegen) {\textbf{3. Generaci\'on de c\'odigo}\\usa intencionalidad interpretada y mensaje original para crear Python};
\node[main=orange, below=of codegen] (security) {\textbf{4. Control de seguridad}\\contrato del script, imports permitidos y salida JSON};
\node[main=orange, below=of security] (execution) {\textbf{5. Ejecuci\'on controlada}\\lanza el script, captura salida y genera artefactos auditables};
\node[main=purple, below=of execution] (interpretation) {\textbf{6. LLM central / interpretaci\'on}\\lee m\'etricas y redacta una conclusi\'on comprensible};
\node[main=gray, below=of interpretation] (output) {\textbf{Resultado final}\\respuesta al usuario, errores y artefactos de ejecuci\'on};

\node[side=red, right=1.15cm of ingest] (invalid) {\textbf{Error de entrada}\\campos incompletos, fechas inv\'alidas o CSV inexistente};
\node[side=red, right=1.15cm of analysis] (llmerror) {\textbf{Error LLM}\\fallo de proveedor, credenciales o formato};
\node[side=red, right=1.15cm of security] (rejected) {\textbf{C\'odigo rechazado}\\imports no permitidos o contrato incumplido};
\node[side=red, right=1.15cm of execution] (execerror) {\textbf{Error de ejecuci\'on}\\fallo del script, \texttt{stderr} o \texttt{returncode}};

\node[statebox, below=0.62cm of output] (state) {\textbf{Estado compartido (\texttt{WorkflowState})}: consulta normalizada, interpretaci\'on estructurada, c\'odigo generado, validaci\'on, salida de ejecuci\'on, avisos y respuesta final.};

\draw[arrow] (input) -- (ingest);
\draw[arrow] (ingest) -- (analysis);
\draw[arrow] (analysis) -- (codegen);
\draw[arrow] (codegen) -- (security);
\draw[arrow] (security) -- (execution);
\draw[arrow] (execution) -- (interpretation);
\draw[arrow] (interpretation) -- (output);

\draw[errorarrow] (ingest.east) -- (invalid.west);
\draw[errorarrow] (analysis.east) -- (llmerror.west);
\draw[errorarrow] (codegen.east) -- ++(0.35,0) |- (llmerror.west);
\draw[errorarrow] (security.east) -- (rejected.west);
\draw[errorarrow] (execution.east) -- (execerror.west);
\draw[errorarrow] (interpretation.east) -- ++(0.35,0) |- (llmerror.west);
\draw[arrow] (invalid.south) |- (output.east);
\draw[arrow] (llmerror.south) |- (output.east);
\draw[arrow] (rejected.south) |- (output.east);
\draw[arrow] (execerror.south) |- (output.east);

\draw[statearrow] (ingest.west) -- ++(-0.35,0) |- (state.west);

\draw[statearrow] (security.east) -- ++(0.45,0) |- (state.east);

\draw[statearrow] (interpretation.east) -- ++(0.45,0) |- (state.east);

] {};
\end{scope}
